cmake_minimum_required(VERSION 3.20)
project(Lab03 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(OpenSSL 3.0 REQUIRED COMPONENTS Crypto)
add_library(labcore STATIC src/task_a.cpp src/task_b.cpp src/task_c.cpp src/task_d.cpp src/task_e.cpp)
target_include_directories(labcore PUBLIC include)
target_link_libraries(labcore PUBLIC OpenSSL::Crypto)
add_executable(lab03 src/main.cpp)
target_link_libraries(lab03 PRIVATE labcore)
if(WIN32 AND MINGW)
  target_link_libraries(labcore PUBLIC crypt32 ws2_32 bcrypt)
endif()
foreach(target labcore lab03)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endforeach()
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 3.9 REQUIRED COMPONENTS Interpreter)
  find_program(OPENSSL_CLI NAMES openssl REQUIRED)
  add_executable(api_checks tests/api_checks.cpp)
  target_link_libraries(api_checks PRIVATE labcore)
  if(NOT MSVC)
    target_compile_options(api_checks PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  add_test(NAME public_api COMMAND api_checks)
  add_test(NAME acceptance COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/run_all.py --exe $<TARGET_FILE:lab03> --openssl ${OPENSSL_CLI})
  set_tests_properties(acceptance PROPERTIES TIMEOUT 600)
endif()
